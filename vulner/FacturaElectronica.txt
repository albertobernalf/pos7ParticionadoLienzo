import xml.etree.ElementTree as ET
from datetime import datetime

def generar_xml_factura():
    # 1. Definición de namespaces UBL 2.1
    NSMAP = {
        'xmlns': 'urn:oasis:names:specification:ubl:schema:xsd:Invoice-2',
        'cac': 'urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents-2',
        'cbc': 'urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2',
        'xsi': 'http://w3.org'
    }

    # Registro de namespaces para ElementTree
    for prefix, uri in NSMAP.items():
        ET.register_namespace(prefix, uri)

    # 2. Raíz del documento
    root = ET.Element('{urn:oasis:names:specification:ubl:schema:xsd:Invoice-2}Invoice')
    
    # --- Datos Cabecera (cbc) ---
    ET.SubElement(root, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}UBLVersionID').text = 'UBL 2.1'
    ET.SubElement(root, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}CustomizationID').text = '10' # Opcional según DIAN
    ET.SubElement(root, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}ProfileID').text = 'DIAN 2.1'
    ET.SubElement(root, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}ID').text = 'SET1' # Número de factura
    ET.SubElement(root, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}UUID', schemeID='1').text = 'Generar CUFE aqui' # CUFE
    ET.SubElement(root, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}IssueDate').text = datetime.now().strftime('%Y-%m-%d')
    ET.SubElement(root, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}IssueTime').text = datetime.now().strftime('%H:%M:%S-05:00')
    ET.SubElement(root, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}InvoiceTypeCode').text = '01' # 01 Factura Venta

    # --- Emisor (AccountingSupplierParty) ---
    supplier = ET.SubElement(root, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents-2}AccountingSupplierParty')
    party = ET.SubElement(supplier, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents-2}Party')
    
    # NIT Emisor
    party_tax = ET.SubElement(party, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents-2}PartyTaxScheme')
    ET.SubElement(party_tax, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}RegistrationName').text = 'EMPRESA EJEMPLO SAS'
    ET.SubElement(party_tax, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}CompanyID', schemeID='6', schemeName='31').text = '900123456' # NIT sin DV
    
    # --- Adquiriente (AccountingCustomerParty) ---
    customer = ET.SubElement(root, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents-2}AccountingCustomerParty')
    customer_party = ET.SubElement(customer, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents-2}Party')
    
    customer_tax = ET.SubElement(customer_party, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents-2}PartyTaxScheme')
    ET.SubElement(customer_tax, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}RegistrationName').text = 'CLIENTE FINAL'
    ET.SubElement(customer_tax, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}CompanyID', schemeID='1', schemeName='13').text = '123456789'

    # --- Ítems (InvoiceLine) ---
    for i in range(1, 3): # Ejemplo con 2 ítems
        line = ET.SubElement(root, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents-2}InvoiceLine')
        ET.SubElement(line, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}ID').text = str(i)
        ET.SubElement(line, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}InvoicedQuantity', unitCode='94').text = '1'
        ET.SubElement(line, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}LineExtensionAmount', currencyID='COP').text = '100000.00'
        
        # Descripción Ítem
        item = ET.SubElement(line, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents-2}Item')
        ET.SubElement(item, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}Description').text = f'Producto o Servicio {i}'
        
        # Precio Ítem
        price = ET.SubElement(line, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents-2}Price')
        ET.SubElement(price, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}PriceAmount', currencyID='COP').text = '100000.00'

    # --- Totales (LegalMonetaryTotal) ---
    total = ET.SubElement(root, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents-2}LegalMonetaryTotal')
    ET.SubElement(total, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}LineExtensionAmount', currencyID='COP').text = '200000.00' # Suma de ítems
    ET.SubElement(total, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}TaxInclusiveAmount', currencyID='COP').text = '238000.00' # Total + IVA
    ET.SubElement(total, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents-2}PayableAmount', currencyID='COP').text = '238000.00' # Total a pagar

    # --- Guardar XML ---
    tree = ET.ElementTree(root)
    tree.write('factura_dian.xml', encoding='UTF-8', xml_declaration=True)
    print("XML generado exitosamente: factura_dian.xml")

def crear_factura_dian():
    # Nodo Raíz
    invoice = ET.Element('{urn:oasis:names:specification:ubl:schema:xsd:Invoice-2}Invoice')
    
    # --- 1. Información General ---
    ET.SubElement(invoice, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}UBLVersionID').text = 'UBL 2.1'
    ET.SubElement(invoice, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}CustomizationID').text = '1'
    ET.SubElement(invoice, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}ID').text = 'SETT123' # Número de factura
    ET.SubElement(invoice, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}IssueDate').text = datetime.now().strftime('%Y-%m-%d')
    ET.SubElement(invoice, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}IssueTime').text = datetime.now().strftime('%H:%M:%S-05:00')
    ET.SubElement(invoice, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}InvoiceTypeCode').text = '01' # Factura de venta
    ET.SubElement(invoice, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}DocumentCurrencyCode').text = 'COP'

    # --- 2. Items de la Factura ---
    items = [
        {"id": "1", "nombre": "Teclado Mecánico", "cantidad": "2.00", "precio": "150000.00", "iva": "19.00"},
        {"id": "2", "nombre": "Mouse Óptico", "cantidad": "1.00", "precio": "80000.00", "iva": "19.00"}
    ]

    subtotal_total = 0
    iva_total = 0

    for i, item in enumerate(items, 1):
        line_id = str(i)
        cant = float(item['cantidad'])
        precio = float(item['precio'])
        iva_rate = float(item['iva'])
        
        base_linea = cant * precio
        impuesto_linea = base_linea * (iva_rate / 100)
        
        subtotal_total += base_linea
        iva_total += impuesto_linea

        # Estructura del Item (InvoiceLine)
        inv_line = ET.SubElement(invoice, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents}InvoiceLine')
        ET.SubElement(inv_line, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}ID').text = line_id

        ET.SubElement(inv_line, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}LineExtensionAmount', currencyID="COP").text = f"{base_linea:.2f}"

        # Impuesto por línea

        ET.SubElement(tax_total, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}TaxAmount', currencyID="COP").text = f"{impuesto_linea:.2f}"
        
        tax_subtotal = ET.SubElement(tax_total, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents}TaxSubtotal')
        ET.SubElement(tax_subtotal, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}TaxableAmount', currencyID="COP").text = f"{base_linea:.2f}"
        ET.SubElement(tax_subtotal, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}TaxAmount', currencyID="COP").text = f"{impuesto_linea:.2f}"
        
        tax_scheme = ET.SubElement(ET.SubElement(tax_subtotal, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents}TaxCategory'), '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents}TaxScheme')
        ET.SubElement(tax_scheme, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}ID').text = "01" # 01 es IVA

        price_node = ET.SubElement(inv_line, '{urn:oasis:names:specification:ubl:schema:xsd:CommonAggregateComponents}Price')

    ET.SubElement(monetary_total, '{urn:oasis:names:specification:ubl:schema:xsd:CommonBasicComponents}PayableAmount', currencyID="COP").text = f"{subtotal_total + iva_total:.2f}"

import os
import zipfile
import uuid
import requests
from lxml import etree
from signxml import XMLSigner, namespaces

# --- CONFIGURACIÓN SET DE PRUEBAS ---
URL_HABILITACION = "https://vpfe-hab.dian.gov.co/WcfDianCustomerServices.svc?wsdl"
TEST_SET_ID = "tu-test-set-id-aqui" # Lo obtienes en el portal de la DIAN

def firmar_xml(xml_string, cert_path, key_path):
    """Firma el XML bajo el estándar XAdES-EPES requerido por la DIAN"""
    root = etree.fromstring(xml_string)
    cert = open(cert_path).read()
    key = open(key_path).read()
    
    # La DIAN requiere que la firma esté dentro de la primera UBLExtension
    signer = XMLSigner(method=namespaces.SignedInfo.ALGORITHM_RSA_SHA256, 
                       signature_algorithm="rsa-sha256", 
                       digest_algorithm="sha256")
    
    signed_root = signer.sign(root, key=key, cert=cert)
    return etree.tostring(signed_root, encoding='utf-8', xml_declaration=True)

def enviar_a_dian(xml_firmado, nit_emisor, invoice_number):
    """Comprime el XML y lo envía vía SOAP SendTestSetAsync"""
    
    # 1. Crear el ZIP en memoria o disco
    zip_filename = f"z{nit_emisor}{invoice_number}.zip"
    with zipfile.ZipFile(zip_filename, 'w') as z:
        z.writestr(f"{invoice_number}.xml", xml_firmado)
    
    with open(zip_filename, "rb") as f:
        zip_base64 = f.read().hex() # La DIAN a veces pide base64 o bytes según el método

    # 2. Construir el Sobre SOAP (Simplificado)
    # Importante: Debes incluir el WS-Security Header con el UsernameToken (SoftwareID)
    soap_body = f"""
    <soap:Envelope xmlns:soap="http://www.w3.org/2003/05/soap-envelope" xmlns:wcf="http://wcf.dian.colombia">
       <soap:Header/>
       <soap:Body>
          <wcf:SendTestSetAsync>
             <wcf:fileName>{zip_filename}</wcf:fileName>
             <wcf:contentFile>{zip_base64}</wcf:contentFile>
             <wcf:testSetId>{TEST_SET_ID}</wcf:testSetId>
          </wcf:SendTestSetAsync>
       </soap:Body>
    </soap:Envelope>
    """
    
    headers = {'Content-Type': 'application/soap+xml;charset=UTF-8'}
    response = requests.post(URL_HABILITACION, data=soap_body, headers=headers)
    
    print(f"Status: {response.status_code}")
    print(f"Respuesta DIAN: {response.text}")
