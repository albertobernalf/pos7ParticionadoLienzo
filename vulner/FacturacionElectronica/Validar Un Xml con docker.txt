python
from lxml import etree
import os

def generar_xml_factura():
    # 1. Crear estructura básica XML (UBL)
    root = etree.Element("Invoice", xmlns="urn:oasis:names:specification:ubl:schema:xsd:Invoice-2")
    etree.SubElement(root, "cbc:ID").text = "SET_PRUEBA_001"
    etree.SubElement(root, "cbc:IssueDate").text = "2026-04-13"
    
    # ... añadir más campos obligatorios ...
    
    tree = etree.ElementTree(root)
    tree.write("factura.xml", pretty_print=True, xml_declaration=True, encoding="UTF-8")
    print("XML generado.")

def validar_xml(xml_path, xsd_path):
    # 2. Validar contra esquema XSD (DIAN/SAT)
    xmlschema_doc = etree.parse(xsd_path)
    xmlschema = etree.XMLSchema(xmlschema_doc)
    
    doc = etree.parse(xml_path)
    if xmlschema.validate(doc):
        print("XML Válido: SI")
    else:
        print("XML Válido: NO")
        print(xmlschema.error_log.filter_from_errors())

if __name__ == "__main__":
    generar_xml_factura()
